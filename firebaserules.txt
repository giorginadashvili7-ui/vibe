rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function validUsername(name) {
      return name is string && name.size() > 0 && name.size() <= 40;
    }

    function isAdmin() {
      return request.auth != null
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    match /users/{userId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.email is string
        && request.resource.data.username is string
        && validUsername(request.resource.data.username)
        && request.resource.data.role == 'user'
        && request.resource.data.createdAt is string;
      allow update: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['username', 'bio', 'profilePicture', 'lastSeen', 'email'])
        && validUsername(request.resource.data.username);
      allow delete: if request.auth != null && (request.auth.uid == userId || isAdmin());
    }

    function validPostData(data) {
      return data.userId is string
        && data.authorName is string
        && data.text is string
        && data.text.size() <= 1000
        && data.mood is string
        && data.mood.size() <= 60
        && data.image is string
        && data.image.size() <= 950000
        && data.reactions is map
        && data.reactions.keys().hasOnly(['love', 'fire'])
        && data.reactions.love is list
        && data.reactions.fire is list
        && data.comments is list
        && data.createdAt is string;
    }

    // Fixed: Removed the unsupported .all() loop. 
    // Rules can only validate the type and size of the array here.
    function commentsAreValid(comments) {
      return comments is list && comments.size() <= 200;
    }

    function validReactions(reactions) {
      return reactions is map
        && reactions.keys().hasOnly(['love', 'fire'])
        && reactions.love is list
        && reactions.fire is list;
    }

    function postContentIsUnchanged() {
      return request.resource.data.userId == resource.data.userId
        && request.resource.data.authorName == resource.data.authorName
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.text == resource.data.text
        && request.resource.data.mood == resource.data.mood
        && request.resource.data.image == resource.data.image;
    }

    match /posts/{postId} {
      allow read: if true;
      allow create: if request.auth != null
        && request.auth.uid == request.resource.data.userId
        && validPostData(request.resource.data)
        && request.resource.data.comments.size() == 0;
        allow update: if request.auth != null
        && postContentIsUnchanged()
        && request.resource.data.keys().hasOnly(['userId', 'authorName', 'createdAt', 'text', 'mood', 'image', 'reactions', 'comments'])
        && request.resource.data.reactions is map
        && request.resource.data.comments is list
        && validReactions(request.resource.data.reactions)
        && commentsAreValid(request.resource.data.comments);
      allow delete: if request.auth != null
        && (request.auth.uid == resource.data.userId || isAdmin());
    }

    match /reports/{reportId} {
      allow read: if request.auth != null && isAdmin();
      allow create: if request.auth != null
        && request.resource.data.reportedBy is string
        && request.resource.data.reason is string
        && request.resource.data.postId is string
        && request.resource.data.createdAt is string;
      allow delete: if request.auth != null && isAdmin();
    }

    match /assets/{assetId} {
      allow read: if true;
      allow write: if request.auth != null && isAdmin();
    }
  }
}
